cmake_minimum_required(VERSION 3.22)
project(active_preempt LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 80 CACHE STRING "Target architecture (A100 = 80)")
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
set(AP_RM_PROFILE "550.120" CACHE STRING "One pinned RM header profile per build: 550.120 or experimental 595.58.03")
set_property(CACHE AP_RM_PROFILE PROPERTY STRINGS "550.120" "595.58.03")
set(NVIDIA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../open-gpu-kernel-modules" CACHE PATH "Unmodified checkout matching AP_RM_PROFILE")
if(AP_RM_PROFILE STREQUAL "550.120")
  set(NVIDIA_EXPECTED_HEAD "5e52edb2034de7db4d8ae368dbc7c26b416bfa16")
  set(AP_RM_EXPERIMENTAL 0)
elseif(AP_RM_PROFILE STREQUAL "595.58.03")
  set(NVIDIA_EXPECTED_HEAD "db0c4e65c8e34c678d745ddb1317f53f90d1072b")
  set(AP_RM_EXPERIMENTAL 1)
else()
  message(FATAL_ERROR "Unreviewed RM profile; CUDA probe must not enable its private ABI")
endif()
execute_process(COMMAND git -C "${NVIDIA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE NVIDIA_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT NVIDIA_HEAD STREQUAL NVIDIA_EXPECTED_HEAD)
  message(FATAL_ERROR "NVIDIA_SOURCE must match the pinned AP_RM_PROFILE commit; do not mix RM ABIs")
endif()
execute_process(COMMAND git -C "${NVIDIA_SOURCE}" diff --quiet HEAD -- src/common/sdk/nvidia/inc src/common/inc src/nvidia/arch/nvalloc/unix/include RESULT_VARIABLE HEADER_DIFF)
if(NOT HEADER_DIFF EQUAL 0)
  message(FATAL_ERROR "Pinned NVIDIA headers have local changes; review them before enabling this ABI")
endif()
configure_file(src/build_profile.h.in generated/build_profile.h @ONLY)
add_library(rm_control SHARED src/rm_control.cpp src/object_registry.cpp src/control_events.cpp src/driver_profile.cpp src/rm_observation.cpp src/group_query_internal.cpp)
target_include_directories(rm_control PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated" "${NVIDIA_SOURCE}/src/common/sdk/nvidia/inc" "${NVIDIA_SOURCE}/src/common/inc" "${NVIDIA_SOURCE}/src/nvidia/arch/nvalloc/unix/include")
target_link_libraries(rm_control PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(rm_control PRIVATE -Wall -Wextra -Werror)
foreach(worker bg_worker int_worker)
  add_executable(${worker} src/${worker}.cu)
  target_link_libraries(${worker} PRIVATE rm_control CUDA::cuda_driver CUDA::cudart Threads::Threads)
  target_compile_options(${worker} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endforeach()
add_executable(preflight_cuda src/preflight.cu)
target_link_libraries(preflight_cuda PRIVATE CUDA::cuda_driver CUDA::cudart ${CMAKE_DL_LIBS})
add_executable(event_dump src/event_dump.cpp)
target_link_libraries(event_dump PRIVATE rm_control)
add_executable(phase2_contract_test src/phase2_contract_test.cpp)
target_link_libraries(phase2_contract_test PRIVATE rm_control)
add_executable(rm_contract_test src/rm_contract_test.cpp)
target_link_libraries(rm_contract_test PRIVATE rm_control)
add_executable(phase3_profile_test src/phase3_profile_test.cpp)
target_link_libraries(phase3_profile_test PRIVATE rm_control)
add_executable(group_binding_test src/group_binding_test.cpp)
target_link_libraries(group_binding_test PRIVATE rm_control)
add_executable(group_preempt_test src/group_preempt_test.cpp)
target_link_libraries(group_preempt_test PRIVATE rm_control)
enable_testing()
add_test(NAME rm_contract_test COMMAND rm_contract_test)
add_test(NAME phase2_contract_test COMMAND phase2_contract_test)
add_test(NAME phase3_profile_test COMMAND phase3_profile_test)
add_test(NAME group_binding_test COMMAND group_binding_test)
add_test(NAME group_preempt_test COMMAND group_preempt_test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME analysis_contract_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_analysis.py")
add_test(NAME runner_contract_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_runner.py")
add_test(NAME schema_integration_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_schema_integration.py" "$<TARGET_FILE:phase2_contract_test>")

add_test(NAME paired_contract_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_pairs.py")
