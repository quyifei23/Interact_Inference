cmake_minimum_required(VERSION 3.22)
project(active_preempt LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 80 CACHE STRING "Target architecture (A100 = 80)")
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
set(NVIDIA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../open-gpu-kernel-modules" CACHE PATH "Unmodified NVIDIA 550.120 checkout")
execute_process(COMMAND git -C "${NVIDIA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE NVIDIA_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT NVIDIA_HEAD STREQUAL "5e52edb2034de7db4d8ae368dbc7c26b416bfa16")
  message(FATAL_ERROR "NVIDIA_SOURCE must be the pinned 550.120 checkout; do not mix RM ABIs")
endif()
add_library(rm_control SHARED src/rm_control.cpp src/object_registry.cpp src/control_events.cpp)
target_include_directories(rm_control PUBLIC src "${NVIDIA_SOURCE}/src/common/sdk/nvidia/inc" "${NVIDIA_SOURCE}/src/common/inc" "${NVIDIA_SOURCE}/src/nvidia/arch/nvalloc/unix/include")
target_link_libraries(rm_control PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(rm_control PRIVATE -Wall -Wextra -Werror)
foreach(worker bg_worker int_worker)
  add_executable(${worker} src/${worker}.cu)
  target_link_libraries(${worker} PRIVATE rm_control CUDA::cuda_driver CUDA::cudart Threads::Threads)
  target_compile_options(${worker} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endforeach()
add_executable(preflight_cuda src/preflight.cu)
target_link_libraries(preflight_cuda PRIVATE CUDA::cuda_driver CUDA::cudart ${CMAKE_DL_LIBS})
add_executable(event_dump src/event_dump.cpp)
target_link_libraries(event_dump PRIVATE rm_control)
add_executable(phase2_contract_test src/phase2_contract_test.cpp)
target_link_libraries(phase2_contract_test PRIVATE rm_control)
add_executable(rm_contract_test src/rm_contract_test.cpp)
target_link_libraries(rm_contract_test PRIVATE rm_control)
enable_testing()
add_test(NAME rm_contract_test COMMAND rm_contract_test)
add_test(NAME phase2_contract_test COMMAND phase2_contract_test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME analysis_contract_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_analysis.py")
add_test(NAME runner_contract_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_runner.py")
add_test(NAME schema_integration_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_schema_integration.py" "$<TARGET_FILE:phase2_contract_test>")
